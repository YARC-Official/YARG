# Curated set of libVLC plugins YARG's background-video playback actually needs.
#
# YARG only plays a single local, muted video file per song -- no network streaming, GUI,
# transcoding, or audio output -- so most of stock libVLC's plugin set (network/stream-out,
# GUI/lua, audio codecs, subtitles, encode-only x264/x265, windowing/vout) is dead weight.
# All three platform jobs in .github/workflows/native-video.yml trim to this list.
#
# One plugin base name per line (matches "foo" in libfoo_plugin.<dylib|dll|so>). If a chart's
# video fails to play after a trim, check the log for a "no suitable decoder/demux module"
# error naming the missing module, then add it back here.

# Local file access + read-ahead caching
filesystem
cache_read
cache_block
attachment

# Containers matching SongEntry.VIDEO_EXTENSIONS (mp4/mov/m4v, avi, mkv/webm, ogv, ts, wmv),
# plus a generic elementary-stream fallback demux
mp4
avi
mkv
ogg
ts
asf
es

# Packetizers the above demuxers depend on for the video codecs below
packetizer_h264
packetizer_hevc
packetizer_mpeg4video
packetizer_mpegvideo
packetizer_av1
packetizer_vc1
packetizer_copy

# Video decode
avcodec
vpx
dav1d
theora
postproc
swscale
h26x

# Hardware-accelerated video decode, per platform (software decode alone can fall behind
# real-time at >1x speed). Each name only exists in one platform's release, so it's a
# harmless "not found" warning in the other two platforms' CI fetch, not an error.
videotoolbox
cvpx
d3d11va
dxva2
vaapi
vdpau

# Chroma conversion (i420/yuv <-> the RV32 format SetVideoFormat requests)
i420_rgb
i420_yuy2
i420_nv12
i420_10_p010
i422_i420
i422_yuy2
yuy2_i420
yuy2_i422
yuvp
grey_yuv
rv32

# Internal filter-chain plumbing used even with no explicit filters configured
chain
scale

# Dummy audio output -- background video audio is muted (see LibVlcVideoPlayer.Mute), but
# libVLC still needs an audio output module to attach to rather than erroring outright
adummy

# The callback-based video output module this project's LibVlcVideoPlayer actually uses
vmem
